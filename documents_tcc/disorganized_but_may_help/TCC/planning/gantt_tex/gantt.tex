    \def\decomposeYearAndDay##1.##2\relax{
        \def\myYear{##1}\def\myDay{##2}
    }
    \decomposeYearAndDay#1\relax
    \pgfcalendardatetojulian{\myYear-01-01}{#2}%
    \advance#2 by-1\relax
    \advance#2 by\myDay\relax
}
